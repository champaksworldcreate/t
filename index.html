3) Flutter (Dart)

// pubspec.yaml: add http: ^1.2.0
import 'dart:convert';
import 'package:http/http.dart' as http;

Future<Map<String, dynamic>> loadChat(String url) async {
  final res = await http.get(Uri.parse(url));
  if (res.statusCode != 200) throw Exception("HTTP ${res.statusCode}");
  return jsonDecode(res.body) as Map<String, dynamic>;
}

// then:
final data = await loadChat("https://.../chat.json");
final messages = (data["messages"] as List).cast<Map<String, dynamic>>();

4) React (Web)

import { useEffect, useState } from "react";

export default function Chat() {
  const [data, setData] = useState(null);

  useEffect(() => {
    fetch("/chat.json").then(r => r.json()).then(setData);
  }, []);

  if (!data) return <div>Loading...</div>;
  return (
    <div>
      {data.messages.map(m => (
        <div key={m.id}>{m.text}</div>
      ))}
    </div>
  );
}

5) React Native

// RN can fetch same JSON. Use FlatList for performance.
import React, { useEffect, useState } from "react";
import { View, Text, FlatList } from "react-native";

export default function Chat() {
  const [data, setData] = useState(null);

  useEffect(() => {
    fetch("https://<user>.github.io/<repo>/chat.json")
      .then(r => r.json())
      .then(setData);
  }, []);

  if (!data) return <Text>Loading...</Text>;

  return (
    <FlatList
      data={data.messages}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <View style={{ padding: 8 }}>
          <Text>{item.text}</Text>
        </View>
      )}
    />
  );
}

6) Python

import json, urllib.request

url = "https://<user>.github.io/<repo>/chat.json"
with urllib.request.urlopen(url) as r:
    data = json.loads(r.read().decode("utf-8"))

print("Participants:", [p["name"] for p in data["participants"]])
print("Total messages:", len(data["messages"]))
